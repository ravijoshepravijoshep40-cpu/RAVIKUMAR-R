# ComicCraft — AI Comic Story Creator

A complete FastAPI + Jinja2 application based on the supplied 23-page ComicCraft specification.

## Workflow

1. User enters story prompt, character name, setting, tone and art style.
2. Gemini Flash generates a structured 5-panel outline.
3. Gemini Pro expands the outline into narration and dialogue.
4. Stable Diffusion generates an illustration for each panel.
5. `layout_builder.py` combines images and story data.
6. `exporters.py` creates a multi-page PDF.
7. The preview page displays the comic and provides a PDF download.
8. The export-success page confirms the workflow.

```powershell
python -m venv env

pip install -r requirements.txt
```


## AI image options

### Option A — Hugging Face API
Set:

```env
HF_API_KEY=...
USE_LOCAL_DIFFUSERS=false
```

### Option B — Local Stable Diffusion
Set:

```env
USE_LOCAL_DIFFUSERS=true
```

A suitable GPU is strongly recommended. The first run downloads the model and can require substantial disk space and VRAM.

### Demo mode

The default `.env.example` enables:

```env
ALLOW_DEMO_FALLBACK=true
```

This means the website can be tested without AI keys. It uses deterministic demo story data and placeholder images. Set it to `false` when you want configuration failures to be reported instead of falling back.

## Important note about model names

The supplied project document specifies `gemini-1.5-flash`, `gemini-1.5-pro`, and `runwayml/stable-diffusion-v1-5`. This implementation keeps those names in `.env` so they can be changed without modifying source code if your provider/account no longer exposes a specified model.

## Security

- Never commit `.env`.
- Do not put API keys into HTML/JavaScript.
- API keys are read server-side only.

from pathlib import Path
from datetime import datetime
import re
import html

from fpdf import FPDF
from PIL import Image

from app.config import BASE_DIR, EXPORTS_DIR

FONT_DIR = BASE_DIR / "assets" / "fonts"

def _find_font():
    candidates = [
        FONT_DIR / "DejaVuSans.ttf",
        Path("/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf"),
        Path("C:/Windows/Fonts/arial.ttf"),
    ]
    for path in candidates:
        if path.exists():
            return path
    return None

def _static_to_file(web_path):
    if web_path.startswith("/static/"):
        return BASE_DIR / "static" / web_path[len("/static/"):]
    return Path(web_path)

def save_pdf(layout, title="ComicCraft Comic"):
    filename = f"comiccraft_{datetime.now().strftime('%Y%m%d_%H%M%S')}.pdf"
    output = EXPORTS_DIR / filename

    pdf = FPDF()
    pdf.set_auto_page_break(auto=True, margin=15)

    font = _find_font()
    if font:
        pdf.add_font("DejaVu", "", str(font), uni=True)
        pdf.add_font("DejaVu", "B", str(font), uni=True)
        regular, bold = "DejaVu", "DejaVu"
    else:
        regular = bold = "Helvetica"

    for panel in layout:
        pdf.add_page()
        pdf.set_font(bold, "B", 18)
        pdf.multi_cell(0, 10, f"Panel {panel['panel_number']}: {panel['title']}")
        pdf.ln(2)

        image_file = _static_to_file(panel["image_path"])
        if image_file.exists():
            try:
                with Image.open(image_file) as im:
                    w, h = im.size
                    max_w, max_h = 180, 100
                    ratio = min(max_w / w, max_h / h)
                    pdf.image(str(image_file), w=w * ratio, h=h * ratio)
                    pdf.ln(5)
            except Exception:
                pass

        pdf.set_font(bold, "B", 11)
        pdf.multi_cell(0, 7, "Scene Description")
        pdf.set_font(regular, "", 10)
        pdf.multi_cell(0, 6, panel.get("scene_description", ""))

        pdf.set_font(bold, "B", 11)
        pdf.multi_cell(0, 7, "Caption")
        pdf.set_font(regular, "", 10)
        pdf.multi_cell(0, 6, panel.get("caption", ""))

        pdf.set_font(bold, "B", 11)
        pdf.multi_cell(0, 7, "Narration")
        pdf.set_font(regular, "", 10)
        pdf.multi_cell(0, 6, panel.get("narration", ""))

        pdf.set_font(bold, "B", 11)
        pdf.multi_cell(0, 7, "Dialogue")
        pdf.set_font(regular, "", 10)
        pdf.multi_cell(0, 6, panel.get("dialogue", ""))

    pdf.output(str(output))
    return f"/static/exports/{output.name}"

import json
import re
from typing import Any

from app.config import GEMINI_API_KEY, GEMINI_FLASH_MODEL, ALLOW_DEMO_FALLBACK, PANELS

def _extract_json(text: str) -> Any:
    text = text.strip()
    if text.startswith("```"):
        text = re.sub(r"^```(?:json)?", "", text.strip(), flags=re.I)
        text = re.sub(r"```$", "", text.strip())
    start = text.find("[")
    end = text.rfind("]")
    if start >= 0 and end > start:
        return json.loads(text[start:end + 1])
    start = text.find("{")
    end = text.rfind("}")
    if start >= 0 and end > start:
        return json.loads(text[start:end + 1])
    raise ValueError("Gemini did not return valid JSON.")

def _demo_outline(story_prompt, character_name, setting, tone, art_style):
    return [
        {
            "panel_number": i,
            "title": title,
            "scene_description": desc,
            "image_prompt": (
                f"{art_style} comic illustration, {setting}, {desc}, "
                f"main character {character_name}, cinematic composition, expressive face, "
                "consistent character design, clean line art, detailed background"
            ),
        }
        for i, (title, desc) in enumerate([
            ("The Beginning", f"{character_name} begins the adventure in {setting}, driven by the idea: {story_prompt}."),
            ("A Strange Discovery", f"{character_name} discovers an unexpected clue in {setting}."),
            ("The Challenge", f"A difficult obstacle appears and tests {character_name}'s courage."),
            ("The Turning Point", f"{character_name} finds a clever way forward and changes the situation."),
            ("A New Beginning", f"The adventure reaches a satisfying ending, leaving {character_name} with hope and a memorable lesson."),
        ], 1)
    ]

def generate_outline(story_prompt, character_name, setting, tone, art_style):
    prompt = f"""
Create exactly {PANELS} connected comic panels for this user request.

Story prompt: {story_prompt}
Main character: {character_name}
Setting: {setting}
Tone: {tone}
Art style: {art_style}

Return ONLY a JSON array. Each object must contain:
panel_number (integer 1-{PANELS}), title, scene_description, image_prompt.
Keep the same main character and visual identity throughout all panels.
The image_prompt must be suitable for an image-generation model and must not contain dialogue text.
"""
    if not GEMINI_API_KEY:
        if ALLOW_DEMO_FALLBACK:
            return _demo_outline(story_prompt, character_name, setting, tone, art_style)
        raise RuntimeError("GEMINI_API_KEY is not configured.")

    try:
        from google import genai
        client = genai.Client(api_key=GEMINI_API_KEY)
        response = client.models.generate_content(
            model=GEMINI_FLASH_MODEL,
            contents=prompt,
            config={"temperature": 0.8, "response_mime_type": "application/json"},
        )
        data = _extract_json(response.text)
        if isinstance(data, dict) and "panels" in data:
            data = data["panels"]
        if not isinstance(data, list) or len(data) < PANELS:
            raise ValueError("Gemini returned an incomplete panel outline.")
        return data[:PANELS]
    except Exception:
        if ALLOW_DEMO_FALLBACK:
            return _demo_outline(story_prompt, character_name, setting, tone, art_style)
        raise

import re
from app.config import GEMINI_API_KEY, GEMINI_PRO_MODEL, ALLOW_DEMO_FALLBACK

def _demo_story(outline, character_name, tone):
    stories = []
    for panel in outline:
        n = panel["panel_number"]
        title = panel["title"]
        desc = panel["scene_description"]
        stories.append({
            "panel_number": n,
            "title": title,
            "scene_description": desc,
            "caption": f"Panel {n}: The story moves forward.",
            "narration": f"{character_name} faces the moment with a {tone} spirit and keeps moving toward the next chapter.",
            "dialogue": f'{character_name}: "I will keep going!"'
        })
    return stories

def _parse_json(text):
    text = re.sub(r"^```(?:json)?", "", text.strip(), flags=re.I)
    text = re.sub(r"```$", "", text.strip())
    import json
    start, end = text.find("["), text.rfind("]")
    if start >= 0 and end > start:
        return json.loads(text[start:end+1])
    raise ValueError("Invalid JSON from Gemini Pro.")

def generate_story(outline, character_name, tone, story_prompt=""):
    prompt = f"""
Expand this {len(outline)}-panel comic outline into a cohesive comic story.

Character: {character_name}
Tone: {tone}
Original prompt: {story_prompt}
Outline:
{outline}

Return ONLY a JSON array. For every panel include:
panel_number, title, scene_description, caption, narration, dialogue.
Keep the story connected and preserve panel order. Dialogue should be short and natural.
"""
    if not GEMINI_API_KEY:
        if ALLOW_DEMO_FALLBACK:
            return _demo_story(outline, character_name, tone)
        raise RuntimeError("GEMINI_API_KEY is not configured.")

    try:
        from google import genai
        client = genai.Client(api_key=GEMINI_API_KEY)
        response = client.models.generate_content(
            model=GEMINI_PRO_MODEL,
            contents=prompt,
            config={"temperature": 0.9, "response_mime_type": "application/json"},
        )
        data = _parse_json(response.text)
        return data
    except Exception:
        if ALLOW_DEMO_FALLBACK:
            return _demo_story(outline, character_name, tone)
        raise

import io
import re
import uuid
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

from app.config import (
    HF_API_KEY, SD_MODEL_ID, USE_LOCAL_DIFFUSERS,
    ALLOW_DEMO_FALLBACK, PANELS_DIR, IMAGE_SIZE
)

_pipeline = None

def _safe_name(prompt: str) -> str:
    slug = re.sub(r"[^a-zA-Z0-9]+", "-", prompt.lower()).strip("-")
    return (slug[:50] or "panel") + "-" + uuid.uuid4().hex[:8] + ".png"

def _demo_image(prompt: str, output: Path):
    # Lightweight offline placeholder so the complete application can run without a GPU/API.
    image = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE), "#f4efe6")
    draw = ImageDraw.Draw(image)
    try:
        font = ImageFont.truetype("DejaVuSans.ttf", 24)
        small = ImageFont.truetype("DejaVuSans.ttf", 16)
    except Exception:
        font = small = None
    title = "COMICCRAFT"
    draw.text((30, 35), title, fill="#111827", font=font)
    wrapped = []
    words = prompt.split()
    line = ""
    for word in words:
        test = (line + " " + word).strip()
        if len(test) > 34:
            wrapped.append(line)
            line = word
        else:
            line = test
    if line:
        wrapped.append(line)
    y = 120
    for line in wrapped[:10]:
        draw.text((30, y), line, fill="#374151", font=small)
        y += 28
    draw.rounded_rectangle((50, 410, IMAGE_SIZE-50, IMAGE_SIZE-50), radius=24, outline="#111827", width=4)
    draw.text((80, 440), "Demo image mode", fill="#111827", font=small)
    image.save(output, "PNG")
    return output

def _local_diffusion(prompt: str, output: Path):
    global _pipeline
    import torch
    from diffusers import StableDiffusionPipeline
    if _pipeline is None:
        dtype = torch.float16 if torch.cuda.is_available() else torch.float32
        _pipeline = StableDiffusionPipeline.from_pretrained(SD_MODEL_ID, torch_dtype=dtype)
        _pipeline = _pipeline.to("cuda" if torch.cuda.is_available() else "cpu")
    result = _pipeline(prompt, num_inference_steps=25, guidance_scale=7.5)
    result.images[0].save(output)
    return output

def _huggingface(prompt: str, output: Path):
    import requests
    if not HF_API_KEY:
        raise RuntimeError("HF_API_KEY is not configured.")
    url = f"https://api-inference.huggingface.co/models/{SD_MODEL_ID}"
    response = requests.post(
        url,
        headers={"Authorization": f"Bearer {HF_API_KEY}"},
        json={"inputs": prompt},
        timeout=180,
    )
    response.raise_for_status()
    output.write_bytes(response.content)
    Image.open(io.BytesIO(response.content)).convert("RGB").save(output)
    return output

def generate_image(prompt: str) -> str:
    output = PANELS_DIR / _safe_name(prompt)
    try:
        if USE_LOCAL_DIFFUSERS:
            _local_diffusion(prompt, output)
        elif HF_API_KEY:
            _huggingface(prompt, output)
        elif ALLOW_DEMO_FALLBACK:
            _demo_image(prompt, output)
        else:
            raise RuntimeError("Configure HF_API_KEY or enable USE_LOCAL_DIFFUSERS.")
    except Exception:
        if ALLOW_DEMO_FALLBACK:
            _demo_image(prompt, output)
        else:
            raise
    return f"/static/panels/{output.name}"

def _story_map(story):
    return {int(item["panel_number"]): item for item in story}

def build_comic_layout(outline, story, image_paths):
    story_by_panel = _story_map(story)
    layout = []
    for index, panel in enumerate(outline):
        number = int(panel["panel_number"])
        details = story_by_panel.get(number, {})
        layout.append({
            "panel_number": number,
            "title": details.get("title") or panel.get("title", f"Panel {number}"),
            "image_path": image_paths[index],
            "scene_description": details.get("scene_description") or panel.get("scene_description", ""),
            "image_prompt": panel.get("image_prompt", ""),
            "caption": details.get("caption", ""),
            "narration": details.get("narration", ""),
            "dialogue": details.get("dialogue", ""),
        })
    return layout

import traceback
from fastapi import APIRouter, Request, Form, HTTPException
from fastapi.responses import HTMLResponse, JSONResponse, RedirectResponse
from fastapi.templating import Jinja2Templates

from app.schemas import PromptRequest, TestImageRequest
from app.gemini_flash import generate_outline
from app.gemini_pro import generate_story
from app.image_generator import generate_image
from app.layout_builder import build_comic_layout
from app.exporters import save_pdf

router = APIRouter()
templates = Jinja2Templates(directory="templates")

def run_generation(data: PromptRequest):
    outline = generate_outline(
        data.story_prompt, data.character_name, data.setting,
        data.tone, data.art_style
    )
    story = generate_story(
        outline, data.character_name, data.tone, data.story_prompt
    )
    image_paths = [
        generate_image(panel.get("image_prompt", panel.get("scene_description", "")))
        for panel in outline
    ]
    layout = build_comic_layout(outline, story, image_paths)
    pdf_path = save_pdf(layout, title=f"{data.character_name}'s Comic")
    return outline, story, layout, pdf_path

@router.get("/", response_class=HTMLResponse)
async def home(request: Request):
    return templates.TemplateResponse("index.html", {"request": request})

@router.post("/generate", response_class=HTMLResponse)
async def generate(
    request: Request,
    story_prompt: str = Form(...),
    character_name: str = Form(...),
    setting: str = Form(...),
    tone: str = Form(...),
    art_style: str = Form(...),
):
    try:
        data = PromptRequest(
            story_prompt=story_prompt,
            character_name=character_name,
            setting=setting,
            tone=tone,
            art_style=art_style,
        )
        outline, story, layout, pdf_path = run_generation(data)
        return templates.TemplateResponse(
            "comic_preview.html",
            {
                "request": request,
                "layout": layout,
                "pdf_path": pdf_path,
                "character_name": data.character_name,
            },
        )
    except Exception as exc:
        traceback.print_exc()
        return templates.TemplateResponse(
            "index.html",
            {"request": request, "error": f"Comic generation failed: {exc}"},
            status_code=500,
        )

@router.post("/generate-comic/json")
async def generate_json(data: PromptRequest):
    try:
        outline, story, layout, pdf_path = run_generation(data)
        return JSONResponse({
            "success": True,
            "panels": layout,
            "pdf_path": pdf_path,
        })
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))

@router.post("/test-image")
async def test_image(data: TestImageRequest):
    try:
        path = generate_image(data.prompt)
        return {"success": True, "image_path": path}
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))

@router.get("/export-success", response_class=HTMLResponse)
async def export_success(request: Request, pdf_path: str = ""):
    return templates.TemplateResponse(
        "export_success.html",
        {"request": request, "pdf_path": pdf_path},
    )

@router.get("/download")
async def download(pdf_path: str):
    # Kept as a compatibility endpoint. Browser can download the static PDF directly.
    if not pdf_path.startswith("/static/exports/"):
        raise HTTPException(status_code=400, detail="Invalid PDF path.")
    return RedirectResponse(url=pdf_path, status_code=303)

from pydantic import BaseModel, Field, field_validator

class PromptRequest(BaseModel):
    story_prompt: str = Field(..., min_length=3, max_length=2000)
    character_name: str = Field(..., min_length=1, max_length=100)
    setting: str = Field(..., min_length=1, max_length=200)
    tone: str = Field(..., min_length=1, max_length=100)
    art_style: str = Field(..., min_length=1, max_length=100)

    @field_validator("*")
    @classmethod
    def strip_values(cls, value):
        if isinstance(value, str):
            value = value.strip()
        return value

class TestImageRequest(BaseModel):
    prompt: str = Field(..., min_length=3, max_length=1000)
